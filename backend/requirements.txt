# HTML sanitization for stored rich text (job descriptions, training
# programme descriptions, announcements). All three are rendered with
# dangerouslySetInnerHTML, and JWTs live in localStorage, so an unsanitized
# payload can steal accounts. nh3 binds the Rust `ammonia` sanitizer: an
# allowlist parse-and-reserialize filter, not a regex over markup.
nh3==0.3.7

# Serves Django Admin's own CSS and JS when DEBUG is false. Django only
# serves them itself in development, so without this the admin renders as
# unstyled HTML behind waitress -- and a separate static server for a handful
# of admin assets is more moving parts than the deployment needs.
whitenoise==6.12.0

# Production WSGI servers -- two, one per platform, and both are used.
#
# `manage.py runserver` is a development server: single-threaded by default,
# no process supervision, and Django's own documentation says not to use it
# for anything resembling production.
#
# Waitress runs on Windows, where this project is developed and where gunicorn
# does not run at all:
#
#     waitress-serve --port=8000 config.wsgi:application
waitress==3.0.2

# Gunicorn serves the Render container, which is Linux. Kept alongside waitress
# rather than replacing it: dropping waitress would leave Windows with no way
# to run the application outside DEBUG, and dropping gunicorn would leave the
# deployment on a server whose own docs call it Windows-oriented. pip installs
# both on both platforms -- gunicorn is importable but unusable on Windows,
# which costs a wheel and avoids a platform marker that silently skips it.
#
#     gunicorn config.wsgi:application --bind 0.0.0.0:$PORT
gunicorn==23.0.0

# S3 API client, for the private document bucket (Cloudflare R2).
#
# Certificates carry identification numbers and free hosting gives a container
# ephemeral storage, so they cannot live on the instance filesystem: the upload
# succeeds, the row is written, and the next deploy discards the file with
# nothing in any log. boto3 rather than django-storages because the requirement
# is four operations on one private bucket -- put, get, delete, list -- and none
# of them is a Django storage backend's job here: reads are streamed back
# through the view that checked ownership, never served by a URL.
boto3==1.40.66
